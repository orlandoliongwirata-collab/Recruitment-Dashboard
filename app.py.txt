import streamlit as st
import pandas as pd
import plotly.express as px

# 1. Setup Tampilan
st.set_page_config(page_title="Recruitment Glow-Up ✨", layout="wide")

st.markdown("""
    <style>
    .main { background-color: #fdf6f9; }
    [data-testid="stMetricValue"] { color: #ff7eb9 !important; font-size: 32px; font-weight: bold; }
    .stButton>button { background-color: #ffb7ce; border-radius: 20px; border: none; color: white; width: 100%; height: 3em; }
    </style>
    """, unsafe_allow_html=True)

# 2. Ambil Data
def load_data():
    sheet_id = "14XHi4b6yzIA_p2AtkgsjQahPeOg16hL4DBAF9OSa39U"
    url = f"https://docs.google.com/spreadsheets/d/{sheet_id}/export?format=csv"
    # Membaca data langsung dari baris 1
    data = pd.read_csv(url)
    return data.dropna(how='all')

try:
    df = load_data()
    
    # Sidebar: Mood & Musik
    with st.sidebar:
        st.title("Team Vibe 💅")
        st.select_slider("How's the energy?", options=["😴", "☕", "🫠", "🔥", "✨"])
        st.divider()
        st.write("🎧 **Focus Beats**")
        st.markdown('<iframe src="https://open.spotify.com/embed/playlist/37i9dQZF1DWZqd5Jicn9Ez" width="100%" height="80" frameBorder="0" allow="autoplay; clipboard-write; encrypted-media; fullscreen; picture-in-picture"></iframe>', unsafe_allow_html=True)

    st.title("Recruitment Squad Dashboard 🌸")
    st.markdown("#### *Let's make magic happen today!* ✨")

    # Ambil Nama Kolom Secara Otomatis
    cols = df.columns.tolist()
    
    # Row 1: Metrics
    c1, c2, c3 = st.columns(3)
    with c1:
        total = df[cols[1]].sum()
        st.metric("Total Hired 🏆", f"{int(total)}")
    with c2:
        avg = df[cols[2]].mean()
        st.metric("Avg. Time to Fill ⏱️", f"{avg:.1f} Days")
    with c3:
        st.metric("Team Happiness 💌", "4.9/5.0")

    st.divider()

    # Row 2: Leaderboard & Grafik
    left, right = st.columns([1, 1.5])
    with left:
        st.subheader("Leaderboard 👑")
        st.dataframe(df, hide_index=True)
        if st.button("Celebrate Wins! 🥳"):
            st.balloons()

    with right:
        st.subheader("Performance Analysis 📈")
        fig = px.bar(df, x=cols[0], y=cols[1], color_discrete_sequence=['#ffb7ce'], template="plotly_white")
        st.plotly_chart(fig, use_container_width=True)

except Exception as e:
    st.error("✨ Menunggu Data Google Sheets... ✨")
    st.info("Tips: Pastikan Baris 1 di Google Sheets adalah Judul (Nama, Hired, Time to Fill) tanpa ada sel yang di-merge.")
