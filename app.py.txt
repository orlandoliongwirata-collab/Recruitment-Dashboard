import streamlit as st

# 1. Konfigurasi Halaman Aesthetic
st.set_page_config(page_title="Recruitment Squad Glow-Up ✨", layout="wide")

# Custom CSS untuk tampilan Pastel & Gen-Z
st.markdown("""
    <style>

    [data-testid="stMetricValue"] { color: #ff7eb9 !important; font-family: 'Inter', sans-serif; }
    .stHeader { background-color: transparent; }

    }

    """, unsafe_allow_html=True)

# 2. Fungsi Ambil Data dari Google Sheets
def load_data():
    # Mengubah link edit menjadi link export CSV agar bisa dibaca Pandas
    sheet_id = "14XHi4b6yzIA_p2AtkgsjQahPeOg16hL4DBAF9OSa39U"
    url = f"https://docs.google.com/spreadsheets/d/{sheet_id}/export?format=csv"
    df = pd.read_csv(url)
    return df

# Load Data
try:
    df = load_data()
    
    # 3. Sidebar: Mood & Music
    with st.sidebar:
        st.title("Team Vibe 💅")
        st.select_slider("How's the energy?", options=["😴", "☕", "🫠", "🔥", "✨"])
        st.divider()
        st.write("🎧 **Focus Beats**")
        # Spotify Lo-fi Playlist
        st.components.v1.iframe("https://open.spotify.com/embed/playlist/37i9dQZF1DWWQRwui0Ex7X?utm_source=generator", height=80)

    # 4. Header & Top Metrics
    st.title("Recruitment Squad Dashboard 🌸")
    st.markdown("#### *Let's hit those targets, besties!* ✨")
    
    col1, col2, col3 = st.columns(3)
    # Sesuaikan nama kolom di bawah dengan nama kolom di Excel Anda
    total_hired = df.iloc[:, 1].sum() # Mengambil kolom ke-2 (Hired)
    avg_ttf = df.iloc[:, 2].mean()   # Mengambil kolom ke-3 (Time to Fill)
    
    col1.metric("Total Hired 🏆", f"{total_hired}", "Monthly Target: 50")
    col2.metric("Avg. Time to Fill ⏱️", f"{avg_ttf:.1f} Days", "-2 days")
    col3.metric("Candidate Love 💌", "4.9/5.0")

    st.divider()

    # 5. Leaderboard & Visualisasi
    c1, c2 = st.columns([1, 2])
    
    with c1:
        st.subheader("Leaderboard 👑")
        # Menampilkan tabel yang lebih cantik
        st.dataframe(df.iloc[:, 0:3], hide_index=True, use_container_width=True)
        if st.button("Celebrate Wins! 🥳"):
            st.balloons()

    with c2:
        st.subheader("Performance Analysis 📈")
        # Grafik batang yang estetik
        fig = px.bar(df, x=df.columns[0], y=df.columns[1], 
                     color_discrete_sequence=['#ffb7ce'],
                     template="plotly_white")
        fig.update_layout(border-radius=20)
        st.plotly_chart(fig, use_container_width=True)

except Exception as e:
    st.error(f"Waduh, koneksi ke Google Sheets gagal! Pastikan link sudah 'Public View'. Error: {e}")
